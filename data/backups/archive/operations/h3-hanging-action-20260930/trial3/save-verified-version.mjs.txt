import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';
import Database from 'better-sqlite3';
const dir='data/backups/h3-hanging-action-20260930/trial3';
assert.equal(JSON.parse(fs.readFileSync(`${dir}/visual-review.json`)).hangingActionPassed,true);
assert.ok(!fs.existsSync(`${dir}/saved-version.json`),'Already imported this verification');
const history=Object.values(JSON.parse(fs.readFileSync(`${dir}/candidate-history.json`)))[0];
assert.equal(history.status.status_str,'success');
const output=history.outputs['14'].images[0];
assert.equal(output.type,'output');
const outputRoot=path.resolve('D:/new_comfyui/output');
const source=path.resolve(outputRoot,output.subfolder,output.filename);
assert.ok(source.startsWith(outputRoot+path.sep));
const graph=JSON.parse(fs.readFileSync(`${dir}/candidate-graph.json`));
assert.deepEqual(history.prompt[2],graph);
const copied=new Database(`${dir}/probe.sqlite`,{readonly:true});
const db=new Database('data/db2.sqlite');
const trackId=1790715576565, projectId=1790665121730;
const before=JSON.parse(fs.readFileSync(`${dir}/prompts-before.json`));
const storyboard=JSON.parse(fs.readFileSync(`${dir}/storyboard-after.json`));
const base=copied.prepare('SELECT prompt FROM o_videoTrack WHERE id=?').get(trackId).prompt;
const variant=copied.prepare('SELECT * FROM o_videoPromptVariant WHERE trackId=? AND language=?').get(trackId,'en-US');
assert.equal(variant.state,'已完成');
assert.equal(variant.prompt,graph['5'].inputs.prompt);
const plans=[...new Set([base,variant.prompt])].map(prompt=>{
  const hash=crypto.createHash('sha256').update(prompt).digest('hex');
  const row=copied.prepare('SELECT * FROM o_h3ReferencePlan WHERE trackId=? AND promptHash=?').get(trackId,hash);
  assert.ok(row,`Missing exact saved reference plan ${hash}`);
  return row;
});
const filePath=`/${projectId}/video/${crypto.randomUUID()}.mp4`;
const target=path.resolve('data/oss'+filePath);
const details=JSON.parse(execFileSync('ffprobe',['-v','error','-select_streams','v:0','-show_entries','stream=codec_name,width,height,r_frame_rate,bit_rate:format=duration,bit_rate','-of','json',source],{encoding:'utf8'}));
const stream=details.streams[0], [num,den]=stream.r_frame_rate.split('/').map(Number);
const previousVideos=db.prepare('SELECT * FROM o_video WHERE videoTrackId=? ORDER BY id').all(trackId);
let videoId;
db.transaction(()=>{
  assert.equal(db.prepare('SELECT videoDesc FROM o_storyboard WHERE id=2').get().videoDesc,storyboard.videoDesc);
  assert.equal(db.prepare('SELECT prompt FROM o_videoTrack WHERE id=?').get(trackId).prompt,before.track.prompt,'Live prompt was edited; do not overwrite');
  assert.deepEqual(db.prepare('SELECT * FROM o_videoPromptVariant WHERE trackId=? ORDER BY language').all(trackId),before.variants.sort((a,b)=>a.language.localeCompare(b.language)),'Language versions changed; do not overwrite');
  for(const plan of plans) for(const slot of JSON.parse(plan.plan).slots) {
    const current=db.prepare('SELECT i.filePath FROM o_assets a LEFT JOIN o_image i ON i.id=a.imageId WHERE a.id=? AND a.projectId=?').get(slot.assetId,projectId);
    assert.equal(current?.filePath,slot.path,'Reference image changed during the probe');
  }
  for(const plan of plans) db.prepare('INSERT INTO o_h3ReferencePlan(trackId,promptHash,plan) VALUES(@trackId,@promptHash,@plan) ON CONFLICT(trackId,promptHash) DO UPDATE SET plan=excluded.plan').run(plan);
  db.prepare('UPDATE o_videoTrack SET prompt=?,state=?,reason=? WHERE id=? AND projectId=?').run(base,'已完成','',trackId,projectId);
  db.prepare('UPDATE o_videoPromptVariant SET prompt=?,state=?,reason=? WHERE trackId=? AND language=?').run(variant.prompt,'已完成','',trackId,'en-US');
  fs.copyFileSync(source,target,fs.constants.COPYFILE_EXCL);
  videoId=Number(db.prepare('INSERT INTO o_video(filePath,time,state,scriptId,projectId,videoTrackId,width,height,fps,bitrate,codec,actualDuration) VALUES(?,?,?,?,?,?,?,?,?,?,?,?)').run(filePath,Date.now(),'生成成功',1,projectId,trackId,stream.width,stream.height,num/den,Number(stream.bit_rate||details.format.bit_rate)||null,stream.codec_name,Number(details.format.duration)).lastInsertRowid);
  db.prepare('INSERT INTO o_videoLanguage(videoId,language,prompt) VALUES(?,?,?)').run(videoId,'en-US',variant.prompt);
})();
assert.deepEqual(db.prepare('SELECT * FROM o_video WHERE videoTrackId=? AND id<>? ORDER BY id').all(trackId,videoId),previousVideos);
assert.equal(db.prepare('SELECT prompt FROM o_videoPromptVariant WHERE trackId=? AND language=?').get(trackId,'en-US').prompt,variant.prompt);
const saved={videoId,filePath,localPath:target,language:'en-US',trackId,comfyPromptId:history.prompt[1],previousVideosPreserved:previousVideos.map(v=>v.id),selectionUnchanged:true};
fs.writeFileSync(`${dir}/saved-version.json`,JSON.stringify(saved,null,2));
console.log(saved);
db.close();copied.close();
